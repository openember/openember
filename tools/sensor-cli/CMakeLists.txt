if(NOT TARGET openember::sensor_config)
    message(FATAL_ERROR "Sensor CLI requires OPENEMBER_COMPONENT_SENSOR and OPENEMBER_COMPONENT_CONFIG")
endif()
find_package(Threads REQUIRED)
add_executable(openember_sensor_cli main.cpp)
target_link_libraries(openember_sensor_cli PRIVATE openember::sensor_config Threads::Threads)
target_compile_features(openember_sensor_cli PRIVATE cxx_std_17)
set_target_properties(openember_sensor_cli PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS openember_sensor_cli RUNTIME DESTINATION bin)
