if(NOT TARGET yaml-cpp::yaml-cpp AND NOT TARGET yaml-cpp)
    message(FATAL_ERROR "OPENEMBER_COMPONENT_CONFIG requires OPENEMBER_WITH_YAMLCPP and a resolved yaml-cpp target")
endif()
add_library(openember_config STATIC src/yaml.cpp)
target_compile_features(openember_config PUBLIC cxx_std_17)
target_include_directories(openember_config PUBLIC include)
if(TARGET yaml-cpp::yaml-cpp)
    target_link_libraries(openember_config PRIVATE yaml-cpp::yaml-cpp)
else()
    target_link_libraries(openember_config PRIVATE yaml-cpp)
endif()
add_library(openember::config ALIAS openember_config)
install(TARGETS openember_config ARCHIVE DESTINATION lib)
install(DIRECTORY include/openember DESTINATION include)
