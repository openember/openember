cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(openember_config_tests LANGUAGES CXX)
    enable_testing()
    set(OPENEMBER_YAMLCPP_LOCAL_SOURCE "" CACHE PATH "Existing yaml-cpp source tree for standalone tests")
    if(OPENEMBER_YAMLCPP_LOCAL_SOURCE)
        set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
        set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
        set(YAML_CPP_BUILD_CONTRIB OFF CACHE BOOL "" FORCE)
        add_subdirectory("${OPENEMBER_YAMLCPP_LOCAL_SOURCE}" yaml-cpp)
    else()
        find_package(yaml-cpp REQUIRED)
    endif()
    add_subdirectory(.. config)
endif()
find_package(Threads REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_executable(test_yaml_config yaml_test.cpp)
target_link_libraries(test_yaml_config PRIVATE openember::config Threads::Threads)
foreach(_case typed defaults structure errors duplicate syntax limits aliases tags files lifetime concurrency)
    add_test(NAME config_yaml_${_case} COMMAND test_yaml_config ${_case} "${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(config_yaml_${_case} PROPERTIES LABELS "unit;regression" TIMEOUT 10)
endforeach()
add_test(NAME config_yaml_export COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/config_export_test.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../scripts/kconfig/genconfig.sh")
set_tests_properties(config_yaml_export PROPERTIES TIMEOUT 20)
