cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(openember_remote_control_tests LANGUAGES CXX)
    enable_testing()
    add_library(remote_control_under_test ../src/gamepad.cpp)
    target_include_directories(remote_control_under_test PUBLIC ../include)
    target_compile_features(remote_control_under_test PUBLIC cxx_std_17)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_sources(remote_control_under_test PRIVATE ../src/f710_evdev.cpp)
        add_executable(openember_gamepad_monitor ../../../tools/gamepad-monitor/main.cpp)
        target_link_libraries(openember_gamepad_monitor PRIVATE remote_control_under_test)
    endif()
    set(_remote_target remote_control_under_test)
else()
    set(_remote_target openember_remote_control)
endif()
add_executable(test_remote_control mapping_test.cpp)
target_link_libraries(test_remote_control PRIVATE ${_remote_target})
foreach(_case neutral edge deadman axes independent_axes stale reconnect invalid config)
    add_test(NAME remote_control_${_case} COMMAND test_remote_control ${_case})
    set_tests_properties(remote_control_${_case} PROPERTIES LABELS "regression;unit" TIMEOUT 10)
endforeach()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(test_remote_evdev evdev_test.cpp)
    target_link_libraries(test_remote_evdev PRIVATE ${_remote_target})
    target_link_options(test_remote_evdev PRIVATE
        -Wl,--wrap=open -Wl,--wrap=close -Wl,--wrap=read -Wl,--wrap=poll -Wl,--wrap=ioctl)
    foreach(_case identity packets buttons initial_x initial_y sync overflow read hangup)
        add_test(NAME remote_evdev_${_case} COMMAND test_remote_evdev ${_case})
        set_tests_properties(remote_evdev_${_case} PROPERTIES LABELS "regression;unit" TIMEOUT 10)
    endforeach()
endif()
