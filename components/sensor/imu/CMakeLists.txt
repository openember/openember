# Assembly is separate from the base interfaces, avoiding a sensor/driver cycle.
add_library(openember_sensor_drivers STATIC src/driver_registry.cpp)
add_library(openember::sensor_drivers ALIAS openember_sensor_drivers)
target_link_libraries(openember_sensor_drivers PUBLIC openember::sensor)
target_compile_features(openember_sensor_drivers PUBLIC cxx_std_17)
if(TARGET openember::yesense_imu)
    target_compile_definitions(openember_sensor_drivers PRIVATE OPENEMBER_IMU_DRIVER_YESENSE=1)
    target_link_libraries(openember_sensor_drivers PRIVATE openember::yesense_imu)
endif()
if(TARGET openember::config)
    add_library(openember_sensor_config STATIC src/configuration_yaml.cpp)
    add_library(openember::sensor_config ALIAS openember_sensor_config)
    target_link_libraries(openember_sensor_config PUBLIC openember::sensor_drivers PRIVATE openember::config)
endif()
