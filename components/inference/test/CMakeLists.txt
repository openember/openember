cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(openember_inference_tests LANGUAGES CXX)
    enable_testing()
    add_subdirectory(.. inference)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_models.py"
    "${CMAKE_CURRENT_BINARY_DIR}/models" RESULT_VARIABLE _model_generation)
if(NOT _model_generation EQUAL 0)
    message(FATAL_ERROR "Cannot generate inference model fixtures")
endif()
add_executable(test_inference_model model_test.cpp)
target_link_libraries(test_inference_model PRIVATE openember::inference)
foreach(_case identity dynamic shape dtype input output missing)
    add_test(NAME inference_${_case} COMMAND test_inference_model "${CMAKE_CURRENT_BINARY_DIR}/models" ${_case})
    set_tests_properties(inference_${_case} PROPERTIES LABELS "regression;unit" TIMEOUT 10)
endforeach()
add_test(NAME inference_config_export COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/config_export_test.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../scripts/kconfig/genconfig.sh")
set_tests_properties(inference_config_export PROPERTIES TIMEOUT 20)
