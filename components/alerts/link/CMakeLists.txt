if(NOT TARGET openember::alerts OR NOT TARGET openember::core OR NOT TARGET openember::msgs_cpp)
    message(FATAL_ERROR "OPENEMBER_COMPONENT_ALERTS_LINK requires ALERTS, LINK and MSGS")
endif()
get_target_property(_alerts_contract openember::msgs_cpp OPENEMBER_ALERTS_CONTRACT_VERSION)
if(NOT _alerts_contract OR _alerts_contract LESS 1)
    message(FATAL_ERROR "alerts Link requires openember-msgs alerts/v1; update the message checkout")
endif()
add_library(openember_alerts_link STATIC src/link_publisher.cpp)
target_compile_features(openember_alerts_link PUBLIC cxx_std_17)
target_include_directories(openember_alerts_link PUBLIC include)
target_link_libraries(openember_alerts_link PUBLIC openember::alerts openember::core)
add_library(openember::alerts_link ALIAS openember_alerts_link)
install(TARGETS openember_alerts_link ARCHIVE DESTINATION lib)
install(DIRECTORY include/openember DESTINATION include)

# Keep the diagnostic target explicitly buildable in products with tools disabled.
set(_monitor_exclude)
if(NOT OPENEMBER_ENABLE_TOOLS)
    set(_monitor_exclude EXCLUDE_FROM_ALL)
endif()
add_executable(openember_alert_monitor ${_monitor_exclude} "${OPENEMBER_SOURCE_DIR}/tools/alert-monitor/main.cpp")
target_link_libraries(openember_alert_monitor PRIVATE openember::alerts_link)
set_target_properties(openember_alert_monitor PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
