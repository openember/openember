project(openember_hardware_interface VERSION 0.1.0 LANGUAGES CXX)

if(NOT OPENEMBER_WITH_YAMLCPP)
    message(FATAL_ERROR "OPENEMBER_SERVICE_HARDWARE_INTERFACE requires OPENEMBER_WITH_YAMLCPP")
endif()

set(HARDWARE_INTERFACE_SRCS
    main.cpp
    src/endpoint.cpp
    src/hardware_config.cpp
    src/hardware_interface_app.cpp
    adapters/sensor_message_adapter.cpp
)
set(HARDWARE_INTERFACE_LIBS
    ${OPENEMBER_LIBRARIES}
    openember::core
    openember_sensor
    openember_hardware
)
set(HARDWARE_INTERFACE_INCLUDES
    ${OPENEMBER_INCLUDE_DIRS}
    ${CMAKE_SOURCE_DIR}/core/include
    ${CMAKE_SOURCE_DIR}/components/hardware/include
    ${CMAKE_SOURCE_DIR}/components/sensor/include
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}
)

if(NOT DEFINED OPENEMBER_COMPONENT_ACTUATOR OR OPENEMBER_COMPONENT_ACTUATOR)
    list(APPEND HARDWARE_INTERFACE_SRCS adapters/actuator_message_adapter.cpp)
    list(APPEND HARDWARE_INTERFACE_LIBS openember_actuator)
    list(APPEND HARDWARE_INTERFACE_INCLUDES
        ${CMAKE_SOURCE_DIR}/components/actuator/include)
endif()

if(NOT DEFINED OPENEMBER_HARDWARE_ENDPOINT_IMU OR OPENEMBER_HARDWARE_ENDPOINT_IMU)
    list(APPEND HARDWARE_INTERFACE_SRCS endpoints/imu_endpoint.cpp)
endif()
if(NOT DEFINED OPENEMBER_HARDWARE_ENDPOINT_TEMPERATURE OR OPENEMBER_HARDWARE_ENDPOINT_TEMPERATURE)
    list(APPEND HARDWARE_INTERFACE_SRCS endpoints/temperature_endpoint.cpp)
endif()
if(NOT DEFINED OPENEMBER_HARDWARE_ENDPOINT_GNSS OR OPENEMBER_HARDWARE_ENDPOINT_GNSS)
    list(APPEND HARDWARE_INTERFACE_SRCS endpoints/gnss_endpoint.cpp)
endif()
if(OPENEMBER_HARDWARE_ENDPOINT_JOINT_CONTROLLER)
    if(DEFINED OPENEMBER_COMPONENT_ACTUATOR AND NOT OPENEMBER_COMPONENT_ACTUATOR)
        message(FATAL_ERROR "OPENEMBER_HARDWARE_ENDPOINT_JOINT_CONTROLLER requires OPENEMBER_COMPONENT_ACTUATOR")
    endif()
    list(APPEND HARDWARE_INTERFACE_SRCS endpoints/joint_controller_endpoint.cpp)
endif()

if(TARGET yaml-cpp::yaml-cpp)
    set(HARDWARE_INTERFACE_YAMLCPP_LIB yaml-cpp::yaml-cpp)
elseif(TARGET yaml-cpp)
    set(HARDWARE_INTERFACE_YAMLCPP_LIB yaml-cpp)
else()
    message(FATAL_ERROR "yaml-cpp target was not found")
endif()

OPENEMBER_ADD_EXECUTABLE(TARGET ${PROJECT_NAME}
    SRCS     ${HARDWARE_INTERFACE_SRCS}
    LIBS     ${HARDWARE_INTERFACE_LIBS}
             ${HARDWARE_INTERFACE_YAMLCPP_LIB}
    INCLUDES ${HARDWARE_INTERFACE_INCLUDES}
    DESTINATION bin
)

set_property(TARGET ${PROJECT_NAME} PROPERTY CXX_STANDARD 17)
set_property(TARGET ${PROJECT_NAME} PROPERTY CXX_STANDARD_REQUIRED ON)

target_compile_definitions(${PROJECT_NAME} PRIVATE
    OPENEMBER_HARDWARE_ENDPOINT_IMU=$<IF:$<BOOL:${OPENEMBER_HARDWARE_ENDPOINT_IMU}>,1,0>
    OPENEMBER_HARDWARE_ENDPOINT_TEMPERATURE=$<IF:$<BOOL:${OPENEMBER_HARDWARE_ENDPOINT_TEMPERATURE}>,1,0>
    OPENEMBER_HARDWARE_ENDPOINT_GNSS=$<IF:$<BOOL:${OPENEMBER_HARDWARE_ENDPOINT_GNSS}>,1,0>
    OPENEMBER_HARDWARE_ENDPOINT_JOINT_CONTROLLER=$<IF:$<BOOL:${OPENEMBER_HARDWARE_ENDPOINT_JOINT_CONTROLLER}>,1,0>
)
